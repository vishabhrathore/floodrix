## Step 2 — Fix the two bugs I flagged in use-execution.ts

**Before you do anything else, open the file you just copied:** `src/features/workflow-canvas/hooks/use-execution.ts`

### Fix A — wrap the render-time `ingest()` call in useEffect

Find this block (around line 140):

```ts
// When poll result comes back with a terminal status, absorb it
if (
  sessionQuery.data &&
  sessionQuery.data.status !== "RUNNING" &&
  pollingRef.current
) {
  ingest(sessionQuery.data);
}
```

Replace it with:

```ts
useEffect(() => {
  if (
    sessionQuery.data &&
    sessionQuery.data.status !== "RUNNING" &&
    pollingRef.current
  ) {
    ingest(sessionQuery.data);
  }
}, [sessionQuery.data, ingest]);
```

`useEffect` is already imported at the top. You're good.

**Add this helper function** somewhere in the file (outside the component):

```ts
function getExecutionHighlightClass(
  status: NodeExecutionStatus | undefined,
  isActive: boolean,
): string {
  if (!status) return "";
  if (isActive && (status === "RUNNING" || status === "WAITING")) {
    return "ring-2 ring-amber-400 ring-offset-2 shadow-lg animate-pulse";
  }
  switch (status) {
    case "RUNNING":
      return "ring-2 ring-amber-300 ring-offset-1 shadow-sm";
    case "WAITING":
      return "ring-2 ring-purple-300 ring-offset-1";
    case "COMPLETED":
      return "ring-1 ring-emerald-300 shadow-sm";
    case "ERRORED":
      return "ring-2 ring-red-400 shadow-md";
    case "SKIPPED":
      return "opacity-50 ring-1 ring-neutral-200";
    default:
      return "";
  }
}
```

**Inside your component function body**, add these lines near the top:

```ts
const executionStatus = useWorkflowCanvasStore(
  (s) => s.nodeExecutionStatus[props.id],
);
const isActive = useWorkflowCanvasStore(
  (s) => s.activeExecutionNodeId === props.id,
);
const executionClass = getExecutionHighlightClass(executionStatus, isActive);
```

**Then find the outer `<div>`** of your node and add `executionClass` to its className:

Before:

```tsx
<div className="rounded-lg border bg-white ...">
```

After:

```tsx
<div className={cn("rounded-lg border bg-white ...", executionClass)}>
```

That's the whole patch. 4 additions, no deletions.
